/**
 * Called by Dashboard.html to fetch all stats from S1
 */
function getDashboardStats() {
  try {
    const ss = getDataSS();
    const branches = ['Goa', 'Lagonoy', 'Tigaon', 'Tinambac', 'Pasacao'];

    // --- Total Accounts ---
    const accountSheet = ss.getSheetByName('Accounts');
    const totalAccounts = accountSheet ? Math.max(0, accountSheet.getLastRow() - 1) : 0;

    // --- Loans Per Branch ---
    const loansPerBranch = {};
    branches.forEach(b => {
      const sheet = ss.getSheetByName(b);
      loansPerBranch[b] = sheet ? Math.max(0, sheet.getLastRow() - 2) : 0;
    });

    // --- Total Payments Collected ---
    const paySheet = ss.getSheetByName('DATA: Payments');
    let totalPayments = 0;
    if (paySheet && paySheet.getLastRow() > 1) {
      const amounts = paySheet.getRange(2, 4, paySheet.getLastRow() - 1, 1).getValues();
      amounts.forEach(r => { totalPayments += parseFloat(r[0]) || 0; });
    }

    // --- Recent Entries (last 8 accounts) ---
    const recentEntries = [];
    if (accountSheet && accountSheet.getLastRow() > 1) {
      const rows = accountSheet.getRange(2, 1, Math.min(8, accountSheet.getLastRow() - 1), 5).getValues();
      rows.forEach(r => {
        recentEntries.push({
          branch: r[0],
          name: r[1] + ', ' + r[2] + (r[3] ? ' ' + r[3] : '') + (r[4] && r[4] !== 'NONE' ? ' ' + r[4] : '')
        });
      });
    }

    return { totalAccounts, loansPerBranch, totalPayments, recentEntries, error: null };
  } catch(e) {
    return { error: e.message };
  }
}

/**
 * Fires when the user clicks a cell in S2.
 * Maps button cells to their functions.
 */
function onSelectionChange(e) {
  const sheet = e.range.getSheet();
  if (sheet.getName() !== 'Dashboard') return;

  const cell = e.range.getA1Notation();

  const actions = {
    'B2': showAccount,
    'D2': showLoanForm,
    'B4': showClientForm,
    'D4': showDashboard,
    'B6': formatExistingDataToCaps,
    'D6': syncBranchDataToConfig
  };

  if (actions[cell]) {
    actions[cell]();
    // Deselect so the same cell can be clicked again
    sheet.getRange('A1').activate();
  }
}

function getClientDetails(id, sheetName) {
  try {
    const ss = getDataSS();
    const sheet = ss.getSheetByName(sheetName);
    const data = sheet.getDataRange().getValues();
    for (let i = 0; i < data.length; i++) {
      if (data[i][0].toString().trim() === id.toString().trim()) {
        const rawRow = data[i].slice(2, 17);
        const cleanValues = rawRow.map((v) => {
          if (v instanceof Date) return Utilities.formatDate(v, "GMT+8", "yyyy-MM-dd");
          return v;
        });
        return { values: cleanValues };
      }
    }
    return { error: "Client ID not found in " + sheetName };
  } catch (e) { return { error: e.message }; }
}

function saveAllPayments(idCode, clientName, allRows, deletedIndices) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000); // Wait up to 30 seconds for other users to finish
    const ss = getDataSS();
    const sheet = ss.getSheetByName("DATA: Payments");
    if (deletedIndices && deletedIndices.length > 0) {
      deletedIndices.sort((a, b) => b - a).forEach(idx => {
        if (idx > 0) sheet.deleteRow(idx);
      });
      SpreadsheetApp.flush();
    }
    allRows.forEach(row => {
      const vals = [idCode, clientName, row.date, row.amount, row.orNum, row.penalty, row.method, "", row.collector, row.notes];
      if (row.rowIdx && row.rowIdx !== "null") {
        const targetRow = parseInt(row.rowIdx);
        if (targetRow <= sheet.getLastRow()) {
          sheet.getRange(targetRow, 1, 1, 10).setValues([vals]);
        }
      } else if (row.date && row.amount > 0) {
        sheet.appendRow(vals);
      }
    });
    SpreadsheetApp.flush();
    return "Successfully synced " + allRows.length + " changes.";
  } catch (e) {
    throw new Error("Sync Failed: " + e.toString());
  } finally {
    lock.releaseLock();
  }
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
    if (!e.postData || !e.postData.contents) throw new Error("No data received");

    const data = JSON.parse(e.postData.contents);
    const ss = getDataSS();

    // CASE A: Loan entry
    if (data.idCode) {
      const targetSheetName = data.branchName || 'Goa';
      const mainSheet = ss.getSheetByName(targetSheetName);
      if (!mainSheet) throw new Error("Sheet '" + targetSheetName + "' not found!");

      const finalId = generateUniqueId(mainSheet, data.idCode);

      let grandTotal = 0;
      data.items.forEach(item => {
        grandTotal += (parseFloat(item.qty) || 0) * (parseFloat(item.unitPrice) || 0);
      });

      const colAValues = mainSheet.getRange(1, 1, mainSheet.getLastRow() + 1, 1).getValues();
      let targetRow = 3;
      for (let i = 2; i < colAValues.length; i++) {
        if (!colAValues[i][0]) { targetRow = i + 1; break; }
        targetRow = i + 2;
      }

      mainSheet.getRange(targetRow, 1).setValue(finalId);
      mainSheet.getRange(targetRow, 2).setValue(data.name.toUpperCase());
      mainSheet.getRange(targetRow, 3).setValue(data.dateReleased);
      mainSheet.getRange(targetRow, 6).setValue(data.frequency);
      mainSheet.getRange(targetRow, 7).setValue(data.terms);
      mainSheet.getRange(targetRow, 10).setValue(grandTotal);
      mainSheet.getRange(targetRow, 13).setValue(data.downpayment);

      const itemSheet = ss.getSheetByName('DATA: Loanitems');
      const itemLookup = ss.getSheetByName('DATA: Items').getDataRange().getValues();
      data.items.forEach(item => {
        let costPrice = 0;
        const match = itemLookup.find(r => r[0].toString().trim().toUpperCase() === item.name.trim().toUpperCase());
        if (match) costPrice = match[2];
        itemSheet.appendRow([finalId, item.qty, item.name.toUpperCase(), costPrice, item.unitPrice, data.crNo, item.serialNo]);
      });

      syncBranchDataToConfig();
      combineAccountData();
      return ContentService.createTextOutput("Loan Success").setMimeType(ContentService.MimeType.TEXT);
    }

    // CASE B: Account entry
    const status = checkDuplicateLive(data.lastName, data.firstName, data.middleName, data.suffix, data.rowIndex);
    if (status === "BLACKLISTED") throw new Error("BLACKLISTED");
    if (status === "DUPLICATE" && (!data.rowIndex || data.rowIndex === "")) throw new Error("DUPLICATE");

    let result = (data.rowIndex && data.rowIndex !== "") ? updateFormInternal(data) : processFormInternal(data);
    return ContentService.createTextOutput("Success: " + result).setMimeType(ContentService.MimeType.TEXT);

  } catch (err) {
    return ContentService.createTextOutput("Error: " + err.message).setMimeType(ContentService.MimeType.TEXT);
  } finally {
    lock.releaseLock();
  }
}

function generateUniqueId(sheet, baseId) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return baseId + "-01";
  const existingIds = sheet.getRange(1, 1, lastRow, 1).getValues().flat();
  let suffix = 1;
  let candidate = baseId + "-" + String(suffix).padStart(2, '0');
  while (existingIds.includes(candidate)) {
    suffix++;
    candidate = baseId + "-" + String(suffix).padStart(2, '0');
  }
  return candidate;
}

function processData(data) {
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(30000);
    const status = checkDuplicateLive(data.lastName, data.firstName, data.middleName, data.suffix, data.rowIndex);
    if (status === "BLACKLISTED") throw new Error("BLACKLISTED");
    if (status === "DUPLICATE" && (!data.rowIndex || data.rowIndex === "")) throw new Error("DUPLICATE");
    let rowResult = (data.rowIndex && data.rowIndex !== "") ? updateFormInternal(data) : processFormInternal(data);
    return "SUCCESS: Row " + rowResult;
  } catch (err) {
    throw new Error(err.message);
  } finally {
    lock.releaseLock();
  }
}

function checkDuplicateLive(lastName, firstName, middleName, suffix, excludeRow = null) {
  const ss = getDataSS();
  const ln = String(lastName  || "").trim().toUpperCase();
  const fn = String(firstName || "").trim().toUpperCase();
  const mn = String(middleName|| "").trim().toUpperCase();
  const sf = String(suffix    || "").trim().toUpperCase();
  if (ln.length < 2 || fn.length < 1) return "";

  const checks = [
    { name: '🚫 Blacklist', type: 'COMBINED', col: 2, status: "BLACKLISTED" },
    { name: 'Accounts', type: 'SPLIT',    col: 2, status: "DUPLICATE"   }
  ];

  for (let s of checks) {
    const sheet = ss.getSheetByName(s.name);
    if (!sheet) continue;
    const lastRow = sheet.getLastRow();
    if (lastRow < 2) continue;

    if (s.type === 'COMBINED') {
      let searchArr = [ln + ", " + fn];
      if (mn) searchArr.push(mn);
      if (sf && sf !== "NONE") searchArr.push(sf);
      const searchString = searchArr.join(" ");
      const blacklistData = sheet.getRange(2, s.col, lastRow - 1, 1).getValues().flat();
      if (blacklistData.some(r => r.toString().toUpperCase().trim() === searchString)) return s.status;
    } else {
      const dataRows = sheet.getRange(2, s.col, lastRow - 1, 4).getValues();
      for (let i = 0; i < dataRows.length; i++) {
        if (excludeRow && (i + 2) == excludeRow) continue;
        if (dataRows[i][0].toString().toUpperCase().trim() === ln &&
            dataRows[i][1].toString().toUpperCase().trim() === fn &&
            dataRows[i][2].toString().toUpperCase().trim() === mn &&
            dataRows[i][3].toString().toUpperCase().trim() === sf) return s.status;
      }
    }
  }
  return "";
}

function searchClient(lastName) {
  const sheet = getDataSS().getSheetByName('Accounts');
  const data = sheet.getDataRange().getValues();
  const ln = lastName.trim().toUpperCase();
  const matches = [];
  for (let i = 1; i < data.length; i++) {

      matches.push({
        rowIndex: i + 1,
        fullName: data[i][1] + ", " + data[i][2] + (data[i][3] ? " " + data[i][3] : ""),
        details: {
          branch: data[i][0], lastName: data[i][1], firstName: data[i][2],

          barangay: data[i][6], city: data[i][7], province: data[i][8],

          telephone: data[i][12], email: data[i][13], email2: data[i][14],
          agent: data[i][15], referred: data[i][16]

}

function getLocationData() {
  try {
    const ss = getDataSS();
    const sheet = ss.getSheetByName('DATA: Location');
    if (!sheet) throw new Error("Sheet 'DATA: Location' not found!");
    const data = sheet.getDataRange().getValues();
    const cities = data[0];
    const locations = {};
    cities.forEach((city, colIndex) => {
      if (!city || city.toString().trim() === "") return;
      const cityName = city.toString().trim();
      const barangays = [];
      for (let rowIndex = 1; rowIndex < data.length; rowIndex++) {
        const brgy = data[rowIndex][colIndex];
        if (brgy && brgy.toString().trim() !== "") barangays.push(brgy.toString().trim());
      }
      locations[cityName] = barangays;
    });
    return locations;
  } catch (err) {
    Logger.log("Error in getLocationData: " + err.message);
    return { "Error": err.message };
  }
}

function formatExistingDataToCaps() {
  const ss = getDataSS();
  const sheet = ss.getSheetByName('Accounts');
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return;
  const range = sheet.getRange(2, 1, lastRow - 1, 16);
  const values = range.getValues();
  const formatted = values.map(row => [
    row[0],
    String(row[1]).toUpperCase(), String(row[2]).toUpperCase(), String(row[3]).toUpperCase(),
    row[4],
    String(row[5]).toUpperCase(), String(row[6]).toUpperCase(), String(row[7]).toUpperCase(),
    String(row[8]).toUpperCase(), String(row[9]).toUpperCase(),
    row[10], row[11], row[12], row[13],
    String(row[14]).toUpperCase(), String(row[15]).toUpperCase()
  ]);
  range.setValues(formatted);
  SpreadsheetApp.getUi().alert("✅ Cleanup Complete: All relevant fields are now ALL CAPS.");
}

function checkExistingId(branchName, baseId) {
  const ss = getDataSS();
  const sheet = ss.getSheetByName(branchName);
  if (!sheet) return { exists: false };
  const lastRow = sheet.getLastRow();
  if (lastRow < 3) return { exists: false };
  const ids = sheet.getRange(3, 1, lastRow - 2, 1).getValues().flat();
  const match = ids.find(id => id.toString().startsWith(baseId));
  return { exists: !!match, existingFullId: match || "", branch: branchName };
}

function syncBranchDataToConfig() {
  const ss = getDataSS(); 
  const configSheet = ss.getSheetByName('DATA: Config');
  const branches = ['Goa', 'Lagonoy', 'Tigaon', 'Tinambac', 'Pasacao'];
  let consolidatedData = [];

  branches.forEach(sheetName => {
    const sheet = ss.getSheetByName(sheetName);
    if (sheet) {
      const lastRow = sheet.getLastRow();
      if (lastRow >= 3) {
        const branchData = sheet.getRange(3, 1, lastRow - 2, 2).getValues();
        branchData.forEach(row => {
          // FIX 1: .toString().trim() removes accidental spacebar presses
          if (String(row[0]).trim() !== "") { 
            consolidatedData.push([row[0], row[1], sheetName]);
          }
        });
      }
    }
  });

  const lastConfigRow = configSheet.getLastRow();
  if (lastConfigRow >= 2) {
    // FIX 2: Subtract 1 from lastConfigRow to prevent invisible out-of-bounds errors
    configSheet.getRange(2, 10, lastConfigRow - 1, 3).clearContent();
  }

  if (consolidatedData.length > 0) {
    configSheet.getRange(2, 10, consolidatedData.length, 3).setValues(consolidatedData);
    SpreadsheetApp.getUi().alert("✅ Sync Complete: " + consolidatedData.length + " records updated.");
  } else {
    SpreadsheetApp.getUi().alert("⚠️ No records found in branch sheets.");
  }
}

function combineAccountData() {
  const ss = getDataSS(); // ← reads/writes to S1

  // 1. Setup Sheet references
  const sourceSheet = ss.getSheetByName("Accounts");
  const destSheet = ss.getSheetByName("DATA: Config");

  // 2. Get the data
  const lastRow = sourceSheet.getLastRow();
  if (lastRow < 2) return;

  const range = sourceSheet.getRange("B2:D" + lastRow);
  const data = range.getValues();

  // 3. Process data into "B, C D." format
  const combinedData = data.map(row => {
    const b = row[0].toString().trim();
    const c = row[1].toString().trim();
    let d = row[2].toString().trim();

    // --- Rule for Column D ---
    if (d.length === 1) {
      d = d + ".";
    } else if (d.includes(" ")) {
      d = d.split(" ")
           .filter(word => word.length > 0)
           .map(word => word[0].toUpperCase() + ".")
           .join("");
    }

    // Combine into final format: B, C D
    let result = b;
    if (b && (c || d)) result += ", ";
    let suffix = (c + " " + d).trim();
    result += suffix;

    return [result];
  }).filter(row => row[0] !== "");

  // 4. Clear and Paste
  destSheet.getRange("I2:I").clearContent();
  if (combinedData.length > 0) {
    destSheet.getRange(2, 9, combinedData.length, 1).setValues(combinedData);
  }
}

// Fetches Items and Prices from the "DATA" sheet
function getInitialData() {
  try {
    // If your script is bound directly to the target spreadsheet, this is safer than openById
    const ss = SpreadsheetApp.getActiveSpreadsheet(); 
    
    // Make sure your sheet is exactly named "DATA" (case-sensitive)
    const dataSheet = ss.getSheetByName('DATA');
    if (!dataSheet) throw new Error("Could not find a sheet named 'DATA'");
    
    const lastRow = dataSheet.getLastRow();
    if (lastRow < 2) return []; // Stop if the sheet only has headers
    
    // Fetch Column A (Item) and Column B (Price)
    const vals = dataSheet.getRange('A2:B' + lastRow).getValues();
    
    return vals
      .filter(r => r[0] !== "") // Ignore empty rows
      .map(r => ({
        item: String(r[0]).trim().toUpperCase(),
        price: Number(r[1]) || 0
      }));
      
  } catch(e) {
    // Return the error so the frontend can display it
    return { error: e.message };
  }
}

// Processes the submitted form
function submitFinancing(payload) {
  const lock = LockService.getScriptLock();
  try {
    // Prevent simultaneous write collisions
    lock.waitLock(10000); 
    const ss = SpreadsheetApp.openById(DB_ID);
    
    // Map dropdown selection to target sheets
    const branchMap = {
      'GOA': '💳 Goa',
      'LAGONOY': '💳 Lagonoy',
      'TIGAON': '💳 Tigaon',
      'TINAMBAC': '💳 Tinambac',
      'PASACAO': '💳 Pasacao'
    };
    
    const targetSheetName = branchMap[payload.branch];
    const sheet = ss.getSheetByName(targetSheetName);
    if (!sheet) throw new Error("Target sheet not found: " + targetSheetName);
    
    const rowsToInsert = [];
    
// Headers sequence:
    // A: Type, B: Date Release, C: Terms, D: Last Name, E: First Name, F: Middle Name, 
    // G: QTY, H: Item, I: Serial No., J: Item Price, K: Total, 
    // L: Grand Total, M: Downpayment, N: Balance, O: OR#, P: [BLANK], Q: OR Date (Paid), R: Remarks
    
    payload.items.forEach((item, index) => {
      const serial = item.serialNo ? item.serialNo.toUpperCase() : ""; // Capture Serial No
      
      if (index === 0) {
        // FIRST ITEM
        rowsToInsert.push([
          payload.type,
          payload.dateRelease,
          payload.terms,
          payload.lastName.toUpperCase(),
          payload.firstName.toUpperCase(),
          payload.middleName.toUpperCase(),
          parseInt(item.qty) || 1,
          item.name.toUpperCase(),
          serial,                     // <--- New Column I
          Number(item.price) || 0,
          Number(item.total) || 0,
          Number(payload.grandTotal) || 0,
          Number(payload.downpayment) || 0,
          Number(payload.balance) || 0,
          payload.orNum,
          "", // Blank column for spacing
          payload.orDate, 
          payload.remarks 
        ]);
      } else {
        // SUBSEQUENT ITEMS: Leave customer details and totals blank.
        rowsToInsert.push([
          "", // A: Type
          "", // B: Date Release
          "", // C: Terms
          "", // D: Last Name
          "", // E: First Name
          "", // F: Middle Name
          parseInt(item.qty) || 1,    // G: QTY
          item.name.toUpperCase(),    // H: Item Name
          serial,                     // I: Serial No.
          Number(item.price) || 0,    // J: Item Price
          Number(item.total) || 0,    // K: Item Total
          "", // L: Grand Total
          "", // M: Downpayment
          "", // N: Balance
          "", // O: OR#
          "", // P: Blank spacer
          "", // Q: Paid Date
          ""  // R: Remarks
        ]);
      }
    });
    
    // Batch insert all item rows at once for speed
    if (rowsToInsert.length > 0) {
      sheet.getRange(sheet.getLastRow() + 1, 1, rowsToInsert.length, rowsToInsert[0].length).setValues(rowsToInsert);
    }
    
    return "SUCCESS";
  } catch(e) {
    throw new Error(e.message);
  } finally {
    lock.releaseLock();
  }
}
