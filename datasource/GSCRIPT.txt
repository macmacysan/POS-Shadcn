// Patched Google Apps Script for POS sync API

// This handles people opening the URL in their web browser (GET requests)
function doGet(e) {
  return ContentService.createTextOutput("The POS API is running successfully. Please use POST to sync data.");
}

// This handles the data sent from your Electron App (POST requests)
function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    var sheetName = payload.sheetName;

    // 1. Handle load action
    if (payload.action === 'load') {
      return loadData(sheetName, payload.query);
    }

    // 2. Handle sync/save action
    var data = payload.data || [];
    if (data.length === 0 && !payload.allowEmpty) {
      return responseJson({ success: false, error: "Empty data payload rejected. Pass allowEmpty:true to explicitly clear." });
    }
    var context = payload.context;

    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName);
    if (!sheet) {
      return responseJson({ success: false, error: "Sheet not found: " + sheetName });
    }

    var DEFAULT_HEADERS = {
      "expenses": ["id", "branch", "syncDate", "type", "description", "receipt", "category", "vat", "amount"],
      "checks": ["id", "branch", "syncDate", "type", "bank", "account", "checkNo", "receipt", "recordDate", "amount"],
      "income": ["id", "branch", "syncDate", "particular", "remarks", "receipt", "recordDate", "amount"],
      "payments": ["id", "branch", "syncDate", "finance", "type", "terms", "date", "accountName", "qty", "item", "unitPrice", "grandTotal", "down", "balance", "cr", "lateFee", "paymentMethod", "notes"],
      "financing": ["id", "branch", "syncDate", "provider", "applicant", "contact", "item", "amount", "terms", "status", "dateApplied"]
    };

    var lastRow = sheet.getLastRow();
    var lastCol = sheet.getLastColumn();
    var normSheetKey = sheetName.toLowerCase();
    var headers = [];

    // Auto-initialize empty sheet with default headers
    if (lastRow < 1 && DEFAULT_HEADERS[normSheetKey]) {
      headers = DEFAULT_HEADERS[normSheetKey];
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
      lastRow = 1;
      lastCol = headers.length;
    } else if (lastRow < 1) {
      return responseJson({ success: false, error: "Sheet has no headers and is not a recognized POS sheet: " + sheetName });
    } else {
      headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
    }

    // Determine id key from context (fall back to 'id') and find column indices (0-based)
    var idKey = (context && context.idKey) ? String(context.idKey).trim() : 'id';
    if (!idKey) idKey = 'id';

    var idColIdx = findHeaderIndex(headers, [idKey]);
    if (idColIdx === -1 && DEFAULT_HEADERS[normSheetKey]) {
      // Auto-fix missing ID column by prepending it
      sheet.insertColumnBefore(1);
      sheet.getRange(1, 1).setValue(idKey);
      lastCol = sheet.getLastColumn();
      headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
      idColIdx = 0;
    } else if (idColIdx === -1) {
      return responseJson({ success: false, error: "'" + idKey + "' column not found in sheet: " + sheetName });
    }

    var branchColIdx = findHeaderIndex(headers, ["branch"]);
    var dateColIdx = findHeaderIndex(headers, ["syncdate", "recorddate", "dateapplied", "date"]);

    // Read all rows currently in the sheet (from row 2 onwards)
    var sheetData = [];
    if (lastRow > 1) {
      sheetData = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();
    }

    // Map key (rowId) -> { rowNumber, branch, syncDate, id }
    var idInfoMap = {};
    for (var r = 0; r < sheetData.length; r++) {
      var rawRowId = String(sheetData[r][idColIdx] || "").trim();
      if (rawRowId !== "") {
        var rowBranch = branchColIdx !== -1 ? String(sheetData[r][branchColIdx] || "").trim() : "";
        var rowDate = "";
        if (dateColIdx !== -1) {
          rowDate = normalizeDateString(sheetData[r][dateColIdx]);
        }

        idInfoMap[rawRowId.toLowerCase()] = {
          rowNumber: r + 2,
          branch: rowBranch,
          syncDate: rowDate,
          id: rawRowId
        };
      }
    }

    // Normalize context
    if (!context && data.length > 0) {
      var firstItem = data[0];
      var firstDate = getItemValue(firstItem, "syncDate") || getItemValue(firstItem, "recordDate") || getItemValue(firstItem, "dateApplied") || getItemValue(firstItem, "date") || "";
      context = {
        branch: String(getItemValue(firstItem, "branch") || "").trim(),
        syncDate: normalizeDateString(firstDate)
      };
    } else if (context) {
      if (context.branch) context.branch = String(context.branch).trim();
      if (context.syncDate) context.syncDate = normalizeDateString(context.syncDate);
      if (context.idKey) context.idKey = String(context.idKey).trim();
    }

    // Extract incoming IDs using configured idKey
    var incomingIds = {};
    for (var k = 0; k < data.length; k++) {
      var idVal = String(getItemValue(data[k], idKey) || "").trim();
      if (idVal !== "") {
        incomingIds[idVal.toLowerCase()] = true;
      }
    }

    // Delete ONLY matching IDKEY that was explicitly removed in the app (via operation: 'delete' in payload)
    var rowsToDelete = [];
    var rowsToAppend = [];

    // Process each item in the payload for updates/appends or explicit deletions
    for (var k = 0; k < data.length; k++) {
      var item = data[k];
      var itemId = String(getItemValue(item, idKey) || "").trim();
      if (itemId === "") continue;

      var operation = getItemValue(item, "operation") || 'upsert';
      var existingInfo = idInfoMap[itemId.toLowerCase()];

      // Bypasses date/branch validation constraints for existing rows in Installment sheets
      if (existingInfo && context && sheetName.indexOf("Installments") !== 0) {
        var branchMatch = !context.branch || existingInfo.branch.toLowerCase() === context.branch.toLowerCase();
        var dateMatch   = !context.syncDate || existingInfo.syncDate === context.syncDate;
        if (!branchMatch || !dateMatch) {
          existingInfo = null; // treat as new row, don't overwrite the other date's record
        }
      }

      if (operation === 'delete') {
        if (existingInfo) {
          if (rowsToDelete.indexOf(existingInfo.rowNumber) === -1) {
            rowsToDelete.push(existingInfo.rowNumber);
          }
        }
        continue;
      }

      // Prepare row values according to sheet headers using case-insensitive getter
      var rowValues = headers.map(function(headerName) {
        var val = getItemValue(item, headerName);
        if (val instanceof Date) {
          return formatToYyyyMmDd(val);
        }
        return val === undefined || val === null ? "" : val;
      });

      if (existingInfo) {
        // Update existing row
        sheet.getRange(existingInfo.rowNumber, 1, 1, headers.length).setValues([rowValues]);
      } else {
        // Queue for append (new row for this branch/date)
        rowsToAppend.push(rowValues);
      }
    }

    // Perform deletions in reverse order to prevent row indices shifting
    rowsToDelete.sort(function(a, b) { return b - a; });
    for (var d = 0; d < rowsToDelete.length; d++) {
      sheet.deleteRow(rowsToDelete[d]);
    }

    // Perform appends
    if (rowsToAppend.length > 0) {
      var nextRow = sheet.getLastRow() + 1;
      sheet.getRange(nextRow, 1, rowsToAppend.length, headers.length).setValues(rowsToAppend);
    }

    return responseJson({
      success: true,
      updated: data.length - rowsToAppend.length - (rowsToDelete.length - (data.filter(function(x) { return getItemValue(x, "operation") === 'delete'; }).length)),
      appended: rowsToAppend.length,
      deleted: rowsToDelete.length
    });

  } catch (error) {
    return responseJson({ success: false, error: error.message });
  }
}

// Filter and load data matching the selected date and branch
function loadData(sheetName, query) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName);
  if (!sheet) {
    return responseJson({ success: false, error: "Sheet not found: " + sheetName });
  }

  var lastRow = sheet.getLastRow();
  var lastCol = sheet.getLastColumn();
  if (lastRow <= 1) {
    return responseJson([]);
  }

  var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
  var rows = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues();

  var branchColIdx = findHeaderIndex(headers, ["branch"]);
  var dateColIdx = findHeaderIndex(headers, ["syncdate", "recorddate", "dateapplied", "date"]);

  var queryBranch = String(query.branch || "").trim().toLowerCase();
  var queryDate = normalizeDateString(query.syncDate);

  var filteredRows = [];
  for (var i = 0; i < rows.length; i++) {
    var row = rows[i];
    
    var shouldInclude = false;
    // Bypasses filtering entirely for Installments accounts & loan sheets
    if (sheetName === "Installments-Accounts" || sheetName.indexOf("Installments") === 0) {
      shouldInclude = true;
    } else {
      var rowBranch = branchColIdx !== -1 ? String(row[branchColIdx] || "").trim().toLowerCase() : "";
      var rowDate = "";
      if (dateColIdx !== -1) {
        rowDate = normalizeDateString(row[dateColIdx]);
      }
      shouldInclude = (rowBranch === queryBranch && rowDate === queryDate);
    }

    if (shouldInclude) {
      var obj = {};
      for (var c = 0; c < headers.length; c++) {
        var cellVal = row[c];
        if (cellVal instanceof Date) {
          cellVal = formatToYyyyMmDd(cellVal);
        }
        obj[headers[c]] = cellVal;
      }
      filteredRows.push(obj);
    }
  }

  return responseJson(filteredRows);
}

// Robust, timezone-independent date formatter to prevent date shifts
function formatToYyyyMmDd(dateObj) {
  if (!(dateObj instanceof Date)) return String(dateObj);
  var y = dateObj.getFullYear();
  var m = String(dateObj.getMonth() + 1);
  var d = String(dateObj.getDate());
  if (m.length === 1) m = "0" + m;
  if (d.length === 1) d = "0" + d;
  return y + "-" + m + "-" + d;
}

// Helper to normalize and parse date objects or date strings cleanly to yyyy-MM-dd
function normalizeDateString(val) {
  if (!val) return "";
  if (val instanceof Date) {
    return formatToYyyyMmDd(val);
  }

  var str = String(val).trim();
  if (str === "") return "";

  // Handle ISO strings or datetime strings
  if (str.indexOf("T") !== -1) {
    str = str.split("T")[0];
  }
  if (str.indexOf(" ") !== -1) {
    str = str.split(" ")[0];
  }

  // Normalize delimiters to dash
  str = str.replace(/\//g, "-");

  var parts = str.split("-");
  if (parts.length === 3) {
    // YYYY-MM-DD
    if (parts[0].length === 4) {
      var y = parts[0];
      var m = parts[1].length === 1 ? "0" + parts[1] : parts[1];
      var d = parts[2].length === 1 ? "0" + parts[2] : parts[2];
      return y + "-" + m + "-" + d;
    }
    // MM-DD-YYYY or DD-MM-YYYY (4-digit third part)
    if (parts[2].length === 4) {
      var parsedDate = new Date(val);
      if (!isNaN(parsedDate.getTime())) {
        return formatToYyyyMmDd(parsedDate);
      }
    }
  }

  return str;
}

// Helper to normalize and match sheet headers case-insensitively and space-insensitively
function findHeaderIndex(headers, targetNames) {
  var normalizedTargets = [];
  for (var i = 0; i < targetNames.length; i++) {
    normalizedTargets.push(targetNames[i].toLowerCase().replace(/[\s\-_]/g, ""));
  }

  for (var i = 0; i < headers.length; i++) {
    var normalizedHeader = String(headers[i] || "").toLowerCase().replace(/[\s\-_]/g, "");
    if (normalizedTargets.indexOf(normalizedHeader) !== -1) {
      return i;
    }
  }
  return -1;
}

// Helper to retrieve value from incoming JSON item using canonical key matching
function getItemValue(item, headerName) {
  if (!item) return "";

  // 1. Try exact match first
  if (item[headerName] !== undefined && item[headerName] !== null) {
    return item[headerName];
  }

  // 2. Match using canonical representation
  var targetCanonical = getCanonicalKey(headerName);
  for (var key in item) {
    if (getCanonicalKey(key) === targetCanonical) {
      return item[key];
    }
  }

  return "";
}

function getCanonicalKey(key) {
  var norm = String(key || "").toLowerCase().replace(/[^a-z0-9]/g, "");
  
  if (norm === "id" || norm === "idkey" || norm === "key") return "id";
  if (norm === "branch" || norm === "branchname") return "branch";
  if (norm === "syncdate" || norm === "recorddate" || norm === "dateapplied" || norm === "date") return "syncdate";
  if (norm === "description" || norm === "particular" || norm === "particulars" || norm === "remarks" || norm === "details" || norm === "desc") return "description";
  if (norm === "receipt" || norm === "receiptno" || norm === "receiptnumber" || norm === "receiptnum" || norm === "ref" || norm === "refno" || norm === "refnum" || norm === "refnumber" || norm === "refid") return "receipt";
  if (norm === "type" || norm === "group") return "type";
  if (norm === "category" || norm === "cat") return "category";
  if (norm === "vat" || norm === "vattype") return "vat";
  if (norm === "amount" || norm === "loanamount" || norm === "grandtotal" || norm === "total") return "amount";
  if (norm === "bank" || norm === "provider" || norm === "finance" || norm === "financeprovider") return "provider";
  if (norm === "account" || norm === "accountname" || norm === "applicantname" || norm === "applicant") return "account";
  if (norm === "checkno" || norm === "checknumber" || norm === "checknum") return "checkno";
  if (norm === "contactnumber" || norm === "contactno" || norm === "contact" || norm === "phone" || norm === "phonenumber") return "contact";
  if (norm === "terms" || norm === "termmonths" || norm === "term") return "terms";
  if (norm === "qty" || norm === "quantity") return "qty";
  if (norm === "unitprice" || norm === "price" || norm === "rate") return "unitprice";
  if (norm === "down" || norm === "downpayment") return "down";
  if (norm === "balance" || norm === "bal") return "balance";
  if (norm === "latefee" || norm === "fee") return "latefee";
  if (norm === "paymentmethod" || norm === "method") return "paymentmethod";
  if (norm === "cr" || norm === "crno" || norm === "crnumber") return "cr";
  if (norm === "notes" || norm === "note") return "notes";
  
  return norm;
}

function responseJson(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
